    <script>
        const { createApp, ref, computed, watch, onMounted, nextTick } = Vue;

        createApp({
            setup() {
                // Application State

                const isDark = ref(false);
                const tasks = ref([]);
                const searchQuery = ref('');
                const selectedCategory = ref('ALL');

                // Modal state
                const showTaskModal = ref(false);
                const editingTaskId = ref(null);
                const formTask = ref({
                    title: '',
                    description: '',
                    isUrgent: true,
                    isImportant: true,
                    priority: 3,
                    estimatedTime: 25,
                    category: 'Lavoro',
                    dueDate: ''
                });

                // Timer State
                const timerSeconds = ref(25 * 60);
                const initialTimerSeconds = ref(25 * 60);
                const isTimerRunning = ref(false);
                const timerInterval = ref(null);
                const activeTimerTask = ref(null);
                const timerMode = ref('work'); // 'work' or 'break'

                // Chart Instances
                let quadrantChartInstance = null;
                let categoryChartInstance = null;

                // Load Initial Data
                onMounted(() => {
                    const savedTasks = localStorage.getItem('pf_tasks');

                        loadSampleTasks();
                    }

                    // System Theme Preference
                    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
                        isDark.value = true;
                        document.documentElement.classList.add('dark');
                    }

                    // Render icons
                    setTimeout(() => lucide.createIcons(), 100);
                });

                // Sample Data initialization
                const loadSampleTasks = () => {

                        {

                            isImportant: true,
                            quadrant: 2,

                            category: 'Lavoro',

                        },
                        {

                            title: 'Rispondere alle email di routine',
                            description: 'Gestione casella di posta in arrivo',
                            isUrgent: true,
                            isImportant: false,
                            quadrant: 3,
                            priority: 2,
                            estimatedTime: 20,
                            timeSpent: 15,
                            category: 'Lavoro',
                            completed: false,
                            dueDate: ''
                        },
                        {
                            id: '4',
                            title: 'Allenamento in palestra',
                            description: 'Sessione di stretching e cardio 40 min',
                            isUrgent: false,
                            isImportant: true,
                            quadrant: 2,
                            priority: 4,
                            estimatedTime: 45,
                            timeSpent: 45,
                            category: 'Salute',
                            completed: true,
                            dueDate: new Date().toISOString().split('T')[0]
                        }
                    ];
                    saveToLocalStorage();
                };

                // Watcher to Sync LocalStorage
                watch(tasks, () => {
                    saveToLocalStorage();
                }, { deep: true });

                watch(activeTab, (newTab) => {
                    nextTick(() => {
                        lucide.createIcons();
                        if (newTab === 'stats') {
                            renderCharts();
                        }
                    });
                });

                const saveToLocalStorage = () => {
                    localStorage.setItem('pf_tasks', JSON.stringify(tasks.value));
                };

                // Smart Priority Index (SPI) Calculator Algorithm
                // Spi Score Formula: (Quadrant Score * 40) + (Priority * 8) + (Due Date Urgency * 20)
                const calculateSPI = (task) => {
                    let quadrantScore = 0;
                    if (task.isUrgent && task.isImportant) quadrantScore = 4; // Q1
                    else if (!task.isUrgent && task.isImportant) quadrantScore = 3; // Q2
                    else if (task.isUrgent && !task.isImportant) quadrantScore = 2; // Q3
                    else quadrantScore = 1; // Q4

                    let dueDateBonus = 0;
                    if (task.dueDate) {
                        const today = new Date().setHours(0,0,0,0);
                        const due = new Date(task.dueDate).setHours(0,0,0,0);
                        const diffDays = Math.ceil((due - today) / (1000 * 60 * 60 * 24));
                        if (diffDays <= 0) dueDateBonus = 25; // Oggi o scaduta
                        else if (diffDays === 1) dueDateBonus = 15; // Domani
                        else if (diffDays <= 3) dueDateBonus = 8;
                    }

                    return (quadrantScore * 25) + (task.priority * 8) + dueDateBonus;
                };

                // Computed Tasks List
                const processedTasks = computed(() => {
                    return tasks.value.map(task => {
                        let q = 4;
                        if (task.isUrgent && task.isImportant) q = 1;
                        else if (!task.isUrgent && task.isImportant) q = 2;
                        else if (task.isUrgent && !task.isImportant) q = 3;

                        return {
                            ...task,
                            quadrant: q,
                            spiScore: calculateSPI(task)
                        };
                    });
                });

                const filteredTasks = computed(() => {
                    return processedTasks.value.filter(task => {
                        const matchesSearch = task.title.toLowerCase().includes(searchQuery.value.toLowerCase()) || 
                                              (task.description && task.description.toLowerCase().includes(searchQuery.value.toLowerCase()));
                        const matchesCat = selectedCategory.value === 'ALL' || task.category === selectedCategory.value;
                        return matchesSearch && matchesCat;
                    });
                });

                const sortedTasks = computed(() => {
                    return [...filteredTasks.value].sort((a, b) => {
                        if (a.completed !== b.completed) return a.completed ? 1 : -1;
                        return b.spiScore - a.spiScore;
                    });
                });

                const topPriorityTask = computed(() => {
                    const uncompleted = sortedTasks.value.filter(t => !t.completed);
                    return uncompleted.length > 0 ? uncompleted[0] : null;
                });

                // Quadrants Helpers
                const getQuadrantTasks = (quadrantNum) => {
                    return processedTasks.value.filter(t => t.quadrant === quadrantNum);
                };

                const getQuadrantBadgeClass = (q) => {
                    switch(q) {
                        case 1: return 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300';
                        case 2: return 'bg-blue-100 text-blue-700 dark:bg-blue-950 dark:text-blue-300';
                        case 3: return 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300';
                        case 4: return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300';
                    }
                };

                // Modal logic
                const openTaskModal = (task = null) => {
                    if (task) {
                        editingTaskId.value = task.id;
                        formTask.value = { ...task };
                    } else {
                        editingTaskId.value = null;
                        formTask.value = {
                            title: '',
                            description: '',
                            isUrgent: true,
                            isImportant: true,
                            priority: 3,
                            estimatedTime: 25,
                            category: 'Lavoro',
                            dueDate: ''
                        };
                    }
                    showTaskModal.value = true;
                    nextTick(() => lucide.createIcons());
                };

                const closeTaskModal = () => {
                    showTaskModal.value = false;
                };

                const saveTask = () => {
                    if (editingTaskId.value) {
                        const index = tasks.value.findIndex(t => t.id === editingTaskId.value);
                        if (index !== -1) {
                            tasks.value[index] = { ...tasks.value[index], ...formTask.value };
                        }
                    } else {
                        tasks.value.push({
                            id: Date.now().toString(),
                            ...formTask.value,
                            timeSpent: 0,
                            completed: false
                        });
                    }
                    closeTaskModal();
                };

                const deleteTask = (id) => {
                    tasks.value = tasks.value.filter(t => t.id !== id);
                    if (activeTimerTask.value && activeTimerTask.value.id === id) {
                        activeTimerTask.value = null;
                        resetTimer();
                    }
                };

                const toggleTaskComplete = (id) => {
                    const task = tasks.value.find(t => t.id === id);
                    if (task) {
                        task.completed = !task.completed;
                    }
                };

                // Timer Logic
                const formattedTime = computed(() => {
                    const m = Math.floor(timerSeconds.value / 60);
                    const s = timerSeconds.value % 60;
                    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
                });

                const startTimerForTask = (task) => {
                    activeTimerTask.value = task;
                    timerMode.value = 'work';
                    setTimerPreset(25, 'work');
                    if (!isTimerRunning.value) {
                        toggleTimer();
                    }
                };

                const setTimerPreset = (minutes, mode = 'work') => {
                    if (isTimerRunning.value) clearInterval(timerInterval.value);
                    isTimerRunning.value = false;
                    timerMode.value = mode;
                    timerSeconds.value = minutes * 60;
                    initialTimerSeconds.value = minutes * 60;
                };

                const toggleTimer = () => {
                    if (isTimerRunning.value) {
                        clearInterval(timerInterval.value);
                        isTimerRunning.value = false;
                    } else {
                        isTimerRunning.value = true;
                        timerInterval.value = setInterval(() => {
                            if (timerSeconds.value > 0) {
                                timerSeconds.value--;
                                // Se è in modalità lavoro e c'è un compito attivo, registra ogni minuto
                                if (timerMode.value === 'work' && activeTimerTask.value) {
                                    if ((initialTimerSeconds.value - timerSeconds.value) % 60 === 0) {
                                        const t = tasks.value.find(x => x.id === activeTimerTask.value.id);
                                        if (t) {
                                            t.timeSpent = (t.timeSpent || 0) + 1;
                                        }
                                    }
                                }
                            } else {
                                clearInterval(timerInterval.value);
                                isTimerRunning.value = false;
                                // Notifica semplice / Feedback
                                alert(timerMode.value === 'work' ? 'Sessione di Focus Completata! Prendi una pausa.' : 'Pausa terminata! Pronto per il prossimo blocco?');
                            }
                        }, 1000);
                    }
                };

                const resetTimer = () => {
                    if (isTimerRunning.value) clearInterval(timerInterval.value);
                    isTimerRunning.value = false;
                    timerSeconds.value = initialTimerSeconds.value;
                };

                // Stats & Analytics Computations
                const completedTasksCount = computed(() => tasks.value.filter(t => t.completed).length);
                
                const completionRate = computed(() => {
                    if (tasks.value.length === 0) return 0;
                    return Math.round((completedTasksCount.value / tasks.value.length) * 100);
                });

                const totalTimeSpent = computed(() => {
                    return tasks.value.reduce((acc, t) => acc + (t.timeSpent || 0), 0);
                });

                const totalMinutesLoggedToday = computed(() => totalTimeSpent.value);

                const importantRatio = computed(() => {
                    const totalImp = processedTasks.value.filter(t => t.isImportant).reduce((acc, t) => acc + (t.timeSpent || 0), 0);
                    if (totalTimeSpent.value === 0) return 0;
                    return Math.round((totalImp / totalTimeSpent.value) * 100);
                });

                const estimationAccuracy = computed(() => {
                    const completed = tasks.value.filter(t => t.completed);
                    if (completed.length === 0) return 100;
                    
                    let totalEst = 0;
                    let totalActual = 0;
                    completed.forEach(t => {
                        totalEst += t.estimatedTime;
                        totalActual += (t.timeSpent || t.estimatedTime);
                    });
                    
                    if (totalEst === 0) return 100;
                    const diff = Math.abs(totalEst - totalActual);
                    const accuracy = Math.max(0, 100 - Math.round((diff / totalEst) * 100));
                    return accuracy;
                });

                // Chart Rendering
                const renderCharts = () => {
                    // Quadrant Time Chart
                    const ctx1 = document.getElementById('quadrantChart');
                    if (ctx1) {
                        if (quadrantChartInstance) quadrantChartInstance.destroy();
                        
                        const qTimes = [1, 2, 3, 4].map(q => {
                            return processedTasks.value.filter(t => t.quadrant === q).reduce((a, b) => a + (b.timeSpent || 0), 0);
                        });

                        quadrantChartInstance = new Chart(ctx1, {
                            type: 'doughnut',
                            data: {
                                labels: ['Q1 (Urgente/Imp)', 'Q2 (Focus/Imp)', 'Q3 (Delega)', 'Q4 (Elimina)'],
                                datasets: [{
                                    data: qTimes,
                                    backgroundColor: ['#ef4444', '#3b82f6', '#f59e0b', '#10b981'],
                                    borderWidth: 0
                                }]
                            },
                            options: {
                                responsive: true,
                                maintainAspectRatio: false,
                                plugins: {
                                    legend: { position: 'bottom' }
                                }
                            }
                        });
                    }

                    // Category Time Chart
                    const ctx2 = document.getElementById('categoryChart');
                    if (ctx2) {
                        if (categoryChartInstance) categoryChartInstance.destroy();

                        const categories = ['Lavoro', 'Personale', 'Studio', 'Salute'];
                        const catTimes = categories.map(cat => {
                            return tasks.value.filter(t => t.category === cat).reduce((a, b) => a + (b.timeSpent || 0), 0);
                        });

                        categoryChartInstance = new Chart(ctx2, {
                            type: 'bar',
                            data: {
                                labels: categories,
                                datasets: [{
                                    label: 'Minuti Registrati',
                                    data: catTimes,
                                    backgroundColor: '#6366f1',
                                    borderRadius: 8
                                }]
                            },
                            options: {
                                responsive: true,
                                maintainAspectRatio: false,
                                plugins: {
                                    legend: { display: false }
                                },
                                scales: {
                                    y: { beginAtZero: true }
                                }
                            }
                        });
                    }
                };

                const toggleDarkMode = () => {
                    isDark.value = !isDark.value;
                    if (isDark.value) {
                        document.documentElement.classList.add('dark');
                    } else {
                        document.documentElement.classList.remove('dark');
                    }
                };

                const formatDate = (dateStr) => {
                    if (!dateStr) return '';
                    const d = new Date(dateStr);
                    return d.toLocaleDateString('it-IT', { day: 'numeric', month: 'short' });
                };

                return {
                    activeTab,
                    isDark,
                    tasks,
                    searchQuery,
                    selectedCategory,
                    showTaskModal,
                    editingTaskId,
                    formTask,
                    timerSeconds,
                    isTimerRunning,
                    activeTimerTask,
                    timerMode,
                    formattedTime,
                    sortedTasks,
                    topPriorityTask,
                    completedTasksCount,
                    completionRate,
                    totalTimeSpent,
                    totalMinutesLoggedToday,
                    importantRatio,
                    estimationAccuracy,
                    getQuadrantTasks,
                    getQuadrantBadgeClass,
                    openTaskModal,
                    closeTaskModal,
                    saveTask,
                    deleteTask,
                    toggleTaskComplete,
                    startTimerForTask,
                    setTimerPreset,
                    toggleTimer,
                    resetTimer,
                    toggleDarkMode,
                    formatDate
                };
            }
        }).mount('#app');
    </script>
